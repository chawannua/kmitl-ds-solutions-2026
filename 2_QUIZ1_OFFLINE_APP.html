ðŸš¨ CRITICAL EXAM TRAP: deque vs list.pop(0)
Never use list.pop(0) for a queue! In Python, list.pop(0) takes O(N) time because all remaining elements must shift in memory. Always use from collections import deque and deque.popleft() which is O(1)!

1. Circular Queue Equations

Circular Queue reuses array slots using modulo arithmetic:

  • Next Rear Index: (rear + 1) % capacity
  • Next Front Index: (front + 1) % capacity
  • Is Empty Condition: size == 0
  • Is Full Condition: size == capacity

2. BFS Grid Template (Search Portal)

from collections import deque

def bfs_grid(grid, start, end):
    rows, cols = len(grid), len(grid[0])
    visited = set([start])  # MARK VISITED BEFORE ENQUEUE!
    q = deque([(start, 0)])
    
    while q:
        (r, c), dist = q.popleft()
        if (r, c) == end:
            return dist
        
        for dr, dc in [(-1,0), (1,0), (0,-1), (0,1)]:
            nr, nc = r + dr, c + dc
            if 0 <= nr < rows and 0 <= nc < cols and grid[nr][nc] != '#' and (nr, nc) not in visited:
                visited.add((nr, nc))  # MARK VISITED IMMEDIATELY
                q.append(((nr, nc), dist + 1))
    return -1
ðŸ’¡ Golden BFS Rule
Always add node to visited BEFORE pushing it to the queue. If you wait until popping, duplicate entries will flood the queue causing TLE/MLE.

class Stack:
    def __init__(self):
        self.items = []

    def push(self, item):
        self.items.append(item)

def infix_to_postfix(expr):
    precedence = {'+': 1, '-': 1, '*': 2, '/': 2, '^': 3}
    right_associative = {'^'}
    output = []
    stack = Stack()

    for ch in expr:
        if ch.isalnum():
            output.append(ch)
        elif ch == '(':
            stack.push(ch)
        elif ch == ')':
            while not stack.is_empty() and stack.peek() != '(':
                output.append(stack.pop())
            stack.pop()
        elif ch in precedence:
            while (not stack.is_empty() and stack.peek() != '(' and
                   (precedence[stack.peek()] > precedence[ch] or
                    (precedence[stack.peek()] == precedence[ch] and ch not in right_associative))):
                output.append(stack.pop())
            stack.push(ch)

    while not stack.is_empty():
        output.append(stack.pop())

    return ''.join(output)


def main():
    expr = input("Enter Infix : ")
    result = infix_to_postfix(expr)
    print(f"Postfix : {result}")

        self.items = []

        self.items.append(item)

        return self.items[-1]

    def is_empty(self):
        return len(self.items) == 0


class Calculator:
    def __init__(self):
        self.stack = Stack()

    def run(self, instructions):
        self.stack = Stack()
        tokens = instructions.split()
        i = 0
        while i < len(tokens):
            token = tokens[i]

            if token == '+':
                x = self.stack.pop()
                y = self.stack.pop()
                self.stack.push(x + y)
            elif token == '-':
                x = self.stack.pop()
                y = self.stack.pop()
                self.stack.push(x - y)
            elif token == '*':
                x = self.stack.pop()
                y = self.stack.pop()
                self.stack.push(x * y)
            elif token == '/':
                x = self.stack.pop()
                y = self.stack.pop()
                self.stack.push(x / y)
            elif token == 'DUP':
                self.stack.push(self.stack.peek())
            elif token == 'POP':
                self.stack.pop()
            elif token == 'PSH':
                i += 1
                self.stack.push(float(tokens[i]))
            else:
                try:
                    num = float(token)
                    self.stack.push(num)
                except ValueError:
                    return f"Invalid instruction: {token}"
            i += 1

        if self.stack.is_empty():
            return 0

        result = self.stack.peek()
        if isinstance(result, float) and result.is_integer():
            result = int(result)
        return result

            return f"car {num} already in soi"

        cur = self.head
        s = str(cur.data)
        while cur.next:

            s += "->" + str(cur.data)
        return s

        if self.isEmpty():

        s = str(cur.data)

            cur = cur.next

        else:
            self.tail.next = new_node

            self.tail = new_node
